let number = 2;
let divisor = 2;
let end = 1000;

let count = 0;

while number <= end do {
    divisor = 2;
    let isPrime = True;

    while divisor * divisor <= number do {
        if number % divisor == 0 then 
            isPrime = False;
        
        divisor = divisor + 1;
    }

    if isPrime then {
        count = count + 1;
        print number;
    }

    number = number + 1;
}

print "Number of primes till " end " is " count;
